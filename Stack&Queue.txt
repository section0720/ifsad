Stl 이용한 스택 사용법
#include <stack>
stack<자료형> stack_name; //스택 선언
stack_name.push(4) //스택 데이터 삽입
stack_name.pop() //스택 top데이터 삭제 (반환값 없음)
stack_name.top() //스택 top데이터 반환 (반환값 존재)
stack_name.size() //스택 크기 반환
stack_name.empty() //스택이 비었는지 확인 (반환값의 자료형은 bool)

priority_queue<자료형> queue_name;  // 최대 힙 기반의 우선 순위 큐 선언
priority_queue<자료형, 컨테이너, 비교연산> queue_name;  // 특정 기준을 삼는 우선 순위 큐 선언, 비교연산에 greater<int>가 들어가면 최소 힙 기반의 우선 순위 큐이고 사용자 정의로 한다면 구조체로 들어감. 컨테이너는 주로 vector임
q.push(6);        // 맨 뒤에 요소 추가
q.emplace(7);     // 맨 뒤에 요소를 직접 생성하여 추가 (임시 객체 생성 방지)
q.pop();          // 맨 앞 요소 제거 (반환값 없음)
q.front();  // 맨 앞 요소 반환 (제거하지 않음)
q.back();    // 맨 뒤 요소 반환 (제거하지 않음)
q.size();     // 큐의 현재 크기 반환
q.empty();  // 큐가 비어있는지 확인
q.top();	//우선 순위 큐에서 가장 앞에 있는 값

{
	bool operator()(int a, int b)
    {

        				// a가 b보다 크면 true를 리턴해 swap할 것
    }
}
